
###############################################################################
# Define arch flags (default to native)
###############################################################################

if (NOT DEFINED ARCH_TYPE)
    if(${CMAKE_SYSTEM_PROCESSOR} STREQUAL "x86_64")
        set(ARCH_TYPE "x86-64-v3")
    endif()
endif()

if (DEFINED ARCH_TYPE)
    set(CMAKE_CXX_ARCH_FLAGS "-march=${ARCH_TYPE}")
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_CXX_COMMON_FLAGS "-Wall -Wextra ${CMAKE_CXX_ARCH_FLAGS} ${CMAKE_CXX_COMMON_FLAGS}")

if (WIN32)
    add_definitions(-DUNICODE -D_UNICODE)
endif()

if(INDEXES_64BITS)
        set(CMAKE_CXX_COMMON_FLAGS "-DINDEXES_64BITS ${CMAKE_CXX_COMMON_FLAGS}")
endif()

if (CODE_COVERAGE)
	set(CODE_COVERAGE_FLAGS "-O0 --coverage")
    set(CTEST_COVERAGE_COMMAND "gcov")
    message(STATUS "Code coverage instrumentation enabled")
endif()

set(CMAKE_CXX_COMMON_FLAGS "${CMAKE_CXX_COMMON_FLAGS} ${CODE_COVERAGE_FLAGS} -fno-finite-math-only")

set(CMAKE_CXX_FLAGS_DEBUG "-ggdb -DDEBUG")

set(CMAKE_CXX_FLAGS_RELEASE "-O3 -ffast-math -DNDEBUG")

set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O3 -ffast-math -DNDEBUG -ggdb")
if (WIN32)
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "${CMAKE_CXX_FLAGS_RELWITHDEBINFO} -s")
endif()
