SET(VERSION_MAJOR 1)
SET(VERSION_MINOR 0)
SET(VERSION_MICRO 0)
set(EMAIL_ADDRESS "jean-baptiste@leonesio.fr")

# Use Component to generate lib and dev package.
set(CPACK_DEB_COMPONENT_INSTALL ON)
set(CPACK_COMPONENTS_ALL lib dev)

INCLUDE(InstallRequiredSystemLibraries)
SET(CPACK_PACKAGE_NAME "squey-utils")
SET(CPACK_PACKAGE_DESCRIPTION_SUMMARY "SQUEY Utils")
SET(CPACK_PACKAGE_VENDOR "${EMAIL_ADDRESS}")
SET(CPACK_PACKAGE_CONTACT "${EMAIL_ADDRESS}")
SET(CPACK_PACKAGE_VERSION_MAJOR ${VERSION_MAJOR})
SET(CPACK_PACKAGE_VERSION_MINOR ${VERSION_MINOR})
SET(CPACK_PACKAGE_VERSION_PATCH ${VERSION_MICRO})
SET(CPACK_STRIP_FILES "TRUE")
SET(CPACK_SOURCE_STRIP_FILES "TRUE")
SET(CPACK_RESOURCE_FILE_LICENSE ${CMAKE_SOURCE_DIR}/LICENSE)
SET(CPACK_SYSTEM_NAME "linux")
SET(CPACK_DEBIAN_PACKAGE_DEPENDS "libhwloc5")

# dev specific stuff
# We can't install header without libs.
set(CPACK_COMPONENT_DEV_DEPENDS lib)

include(CPack)
